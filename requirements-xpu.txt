# Install with uv (pip has no --override; pyzes needs one to beat torch's pyzes==0.1.1):
#   grep -v '^transformers' requirements.txt > /tmp/requirements-core.txt
#   echo 'pyzes>=0.1.2' > /tmp/override-pyzes.txt
#   uv --no-config pip install -r /tmp/requirements-core.txt -r requirements-xpu.txt --override /tmp/override-pyzes.txt
# --no-config matters: run inside a verl checkout, uv reads its pyproject [tool.uv].override-dependencies
# (transformers==5.12.1, kernels<0.15 for the CUDA stack), which would silently beat the pins below.

kernels>=0.16.0,<0.17.0
dpctl
